\subsection{Streaming Scoring and Calibration Pseudocode}
\begin{algorithm}[t]
\caption{Streaming Anomaly Scoring and Decision Calibration}
\label{alg:scoring_protocol}
\begin{algorithmic}[1]
\Require Streaming telemetry $\mathbf{X} \in \mathbb{R}^{T \times K}$, filtration $\mathcal{F}_t = \sigma(\mathbf{X}[1:t, :])$, context horizon $C=256$, suspect horizon $S=64$, deployment mode $\mathcal{M} \in \{\text{Causal}, \text{Buffered}\}$, calibrator $\mathcal{C}_{\text{EVT}}$ fit to nominal residuals.
\Ensure Point-level alarm stream $\hat{y}_t \in \{0, 1\}$ for $t \in [C, T]$.
\State Initialize score buffer $\mathbf{s} \in \mathbb{R}^T \leftarrow \mathbf{0}$, hit counts $\mathbf{c} \in \mathbb{R}^T \leftarrow \mathbf{0}$.
\For{$t = C$ \textbf{to} $T - S$}
    \State Extract historical context: $\mathbf{x}_{\text{ctx}} \leftarrow \mathbf{X}[t - C : t, :]$ \Comment{$\mathcal{F}_t$-measurable}
    \State Encode latent context: $z_{\text{ctx}} \leftarrow E_\theta(\mathbf{x}_{\text{ctx}})$
    \State Forecast future latent representation: $\hat{z}_{\text{tgt}} \leftarrow P_\psi(z_{\text{ctx}})$
    \If{$\mathcal{M} = \text{Causal}$ \textbf{($0$-Lookahead Delay)}}
        \State Target is trailing window: $\mathbf{x}_{\text{tgt}}^{\text{causal}} \leftarrow \mathbf{X}[t - S : t, :]$ \Comment{Strictly $\mathcal{F}_t$-measurable}
        \State Encode target: $z_{\text{tgt}} \leftarrow E_{\bar{\theta}}(\mathbf{x}_{\text{tgt}}^{\text{causal}})$
        \State Compute trailing predictive discrepancy: $s_t \leftarrow \mathcal{D}_{\text{sal}}(\hat{z}_{\text{tgt}} - z_{\text{tgt}})$
        \State $\hat{y}_t \leftarrow \mathbb{I}(s_t > \tau_{\text{eff}})$ \Comment{Instantaneous decision at timestamp $t$}
    \ElsIf{$\mathcal{M} = \text{Buffered}$ \textbf{($S$-Lookahead Delay)}}
        \State Extract prospective suspect horizon: $\mathbf{x}_{\text{tgt}} \leftarrow \mathbf{X}[t : t + S, :]$ \Comment{$\mathcal{F}_{t+S}$-measurable}
        \State Encode target representation: $z_{\text{tgt}} \leftarrow E_{\bar{\theta}}(\mathbf{x}_{\text{tgt}})$
        \State Compute forward predictive discrepancy: $\mathcal{D}_t \leftarrow \mathcal{D}_{\text{sal}}(\hat{z}_{\text{tgt}} - z_{\text{tgt}})$
        \For{$\tau = t$ \textbf{to} $t + S - 1$}
            \State Accumulate suspect overlap: $s_\tau \leftarrow s_\tau + \mathcal{D}_t$, \quad $c_\tau \leftarrow c_\tau + 1$
        \EndFor
        \State Finalize running average score at step $t$: $\bar{s}_t \leftarrow s_t / \max(c_t, 1)$
        \State $\hat{y}_t \leftarrow \mathbb{I}(\bar{s}_t > \tau_{\text{eff}})$ \Comment{Finalized at timestamp $t+S$}
    \EndIf
\EndFor
\State \Return Binary alarm sequence $\hat{\mathbf{y}}$
\end{algorithmic}
\end{algorithm}
