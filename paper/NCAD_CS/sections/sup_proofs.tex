\section{Proofs of Propositions}
\label{sup:proofs}

\begin{proof}[Proof of Proposition~1]
\textbf{(Operator Non-Triviality and Barrier Behavior).}
(i) At $\mathbf{K} = \mathbf{0}$, all eigenvalues $\lambda_i = 0$, so the smoothed distribution $p_i = \epsilon_{\log}/(D\epsilon_{\log}) = 1/D$ yields $\Delta\mathcal{H}(\mathbf{0}) = 0$, while the smooth shifted energy barrier evaluates to $\Phi_{\text{energy}}(\mathbf{0}) = -1 - \log\epsilon + \log(1 + \epsilon) \approx 12.82$ with strictly negative radial derivative $\partial\Phi_{\text{energy}}/\partial r\big|_{r=0} = 1 - 1/\epsilon < 0$. Simultaneously, the prediction term has gradient $\nabla_{\mathbf{K}}\mathcal{L}_{\text{pred}}\big|_{\mathbf{0}} = -\frac{2}{BD}\sum_{b=1}^B z_{\text{tgt}}^{(b)} {z_{\text{ctx}}^{(b)}}^T \neq \mathbf{0}$ by the coupling assumption. Hence $\mathbf{K} = \mathbf{0}$ is not a stationary point and cannot be a local minimizer, while the shifted penalty at the origin remains finite, $\mathcal{R}_{\text{spectral}}(\mathbf{0}) = \beta_{\text{norm}}(-1 - \log\epsilon + \log(1+\epsilon)) \approx 12.8\,\beta_{\text{norm}}$. (ii) In the unshifted limit ($\epsilon \to 0^+$), $-\log r \to +\infty$ as $\|\mathbf{K}\|_F \to 0^+$ while $\mathcal{L}_{\text{pred}} \to E_{\text{tgt}} < \infty$, so $\mathcal{L}(\mathbf{K}) \to +\infty$; the identity operator satisfies $\Delta\mathcal{H}(\mathbf{I}) = \Phi_{\text{energy}}(\mathbf{I}) = 0$ with strictly finite loss, so $\mathbf{K} = \mathbf{0}$ cannot be a global minimizer in the idealization either.
\end{proof}

\begin{proof}[Proof of Proposition~2]
\textbf{(Idealized Continuous Gradient Flow Motivation: Absence of Periodic Orbits).}
Suppose $\gamma(t)$ is a periodic orbit with period $T_p > 0$, i.e.\ $\gamma(t + T_p) = \gamma(t)$. Integrating the time derivative of $\Phi$ along the closed path yields $\int_0^{T_p} \dot\Phi(\gamma(t))\, dt = 0$, since $\Phi(\gamma(T_p)) = \Phi(\gamma(0))$. By the chain rule, $\dot\Phi(\gamma(t)) = \langle \nabla\Phi, \dot{\gamma} \rangle$, which equals $-\|\nabla\Phi(\gamma(t))\|_2^2 \le 0$. The integral of a non-positive integrand vanishes only if $\|\nabla\Phi(\gamma(t))\|_2 \equiv 0$ on $[0, T_p]$, hence $\dot{\gamma}(t) \equiv \mathbf{0}$: $\gamma$ is an isolated fixed point, contradicting a non-constant periodic orbit.
\end{proof}
