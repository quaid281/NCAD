\subsection{Stage 1: Multiscale Windowing, Multi-Depth TCN Hierarchy Tapping, and Barycenter Centering}
\label{subsec:architecture}
\label{subsec:shared_inductive_biases}
Consider a multivariate telemetry stream $\mathbf{X} \in \mathbb{R}^{T \times K}$ recording $K$ sensor channels over $T$ timesteps. Fig.~\ref{fig:triad_architecture} illustrates the four-stage pipeline from raw telemetry to binary alarm decisions. In Stage~1, a sliding window of length $W = C + S = 320$ advances at stride $1$ and splits the stream at index $t$ into a historical \emph{context window} $\mathbf{x}_{\text{ctx}} = \mathbf{X}[t - C : t, :] \in \mathbb{R}^{C \times K}$ ($C = 256$) and a future \emph{target horizon} $\mathbf{x}_{\text{tgt}} = \mathbf{X}[t : t + S, :] \in \mathbb{R}^{S \times K}$ ($S = 64$, up to causal decision timestamp $\tau = t + S - 1$). To separate slow drift from high-frequency vibration before cross-channel mixing, multichannel inputs ($K > 1$) pass through a causal depthwise 1D convolutional stem ($K \le 1$ uses an identity bypass) and are split across $J=4$ smooth dyadic shells on a $256$-point real fast Fourier transform (rFFT) grid normalized into an amplitude partition of unity ($\sum_{j=1}^J \Psi_j[\omega_m] \equiv 1$ within $1.2\times 10^{-7}$), filtered by per-shell causal 1D convolutions (kernels $\{15, 9, 5, 3\}$), and recombined via a softmax router into a causal dilated temporal convolutional network (TCN) \cite{bai2018empirical}.

Unlike reconstructive detectors that penalize raw waveform error $\|\mathbf{X} - \hat{\mathbf{X}}\|_F^2$, we map windows into a compact 32-dimensional latent space ($D=32$) using a Joint-Embedding Predictive Architecture (JEPA) \cite{lecun2022path}. Dual online ($E_\theta$) and target ($E_\phi$) encoders map $\mathbf{x}_{\text{ctx}}$ and $\mathbf{x}_{\text{tgt}}$ to approximately radius-$\sqrt D$ layer-normalized vectors; the normalization epsilon prevents exact sphere membership. Target weights $\phi$ track online weights $\theta$ via an exponential moving average (EMA), $\phi_k \leftarrow m \phi_{k-1} + (1 - m) \theta_k$ with momentum $m = 0.996$ ($0.995$ in archive cohorts) \cite{grill2020bootstrap, assran2023self}. Because deep dilated TCN stacks can smooth sharp onset spikes while non-stationary regimes shift the latent mean, Stage~1 introduces two operations inside both encoders:
\begin{enumerate}[leftmargin=*,nosep,itemsep=1pt]
    \item \emph{Multi-depth functional hierarchy tapping:} To preserve fast transient spikes alongside long-range context, a tap head extracts $L_{\text{tap}}=3$ feature maps $\mathbf{H}^{(\ell)} \in \mathbb{R}^{B \times T_w \times d_{\text{model}}}$ ($d_{\text{model}}=48$) from the shallow dyadic stem, mid-depth TCN block, and terminal TCN block. Concatenating the channel-centered terminal state and temporal root-mean-square (RMS) energy of each tap forms $\mathbf{h}_{\text{tap}} \in \mathbb{R}^{B \times 6 d_{\text{model}}}$, which is projected to $\mathbb{R}^D$ and fused with the deep representation $\mathbf{z}_{\text{deep}}$ via a bounded gate $g = 0.5\,\sigma(\theta_g) \le 0.5$:
    \begin{equation}
        \mathbf{z}_{\text{fused}} = (1 - g)\,\mathbf{z}_{\text{deep}} + g\,\operatorname{LN}\bigl(W_2\,\phi_{\text{GELU}}(\operatorname{LN}(W_1 \mathbf{h}_{\text{tap}}))\bigr).
        \label{eq:multidepth_tap}
    \end{equation}
    \item \emph{Synchronized barycenter centering:} An online running centroid $\bar{\bm{\mu}}_{\text{ctx}} \in \mathbb{R}^D$ (updated at rate $0.05$ on training batches) is copied synchronously to the EMA target encoder and subtracted from both context and target vectors prior to terminal LayerNorm, $z = \operatorname{LN}(\mathbf{z}_{\text{fused}} - \bar{\bm{\mu}}_{\text{ctx}})$. This attenuates learned common-mode offsets without guaranteeing invariance to unseen drift.
\end{enumerate}

\subsection{Stage 2: Nested Causal Multi-Horizon Rollout and Variance-Floor Decorrelation Objective}
\label{subsec:triad_formulations}
In the single-horizon baseline (\emph{TS-JEPA}), a two-layer multi-layer perceptron (MLP) predictor $P_\theta: \mathbb{R}^D \to \mathbb{R}^D$ maps $z_{\text{ctx}}$ to a full-window forecast $\hat{z}_{\text{tgt}} = P_\theta(z_{\text{ctx}})$, trained with predictive mean-squared error (MSE) and a log-determinant volume term $-\frac{\beta}{D}\ln\det(\mathbf{\Sigma} + \epsilon\mathbf{I})$ ($\beta = 0.05$) related to VICReg \cite{bardes2021vicreg}. LayerNorm limits each vector's scale but does not bound its temporal tail kurtosis; pooling an $S=64$ window into one vector can average out brief faults.

To resolve how a trajectory evolves across the target window, Stage~2 of \emph{Laminar-JEPA} uses $M=4$ nested prefixes $\mathbf{x}_{\text{tgt}}^{(m)} = \mathbf{x}_{\text{tgt}}[0 : S_m, :]$ with $S_m=\min(S,\max(2,\lfloor mS/M\rfloor))$ for $m<M$ and $S_M=S$ ($\{16,32,48,64\}$ at $S=64$). Each is encoded by $E_\phi$ into $z_{\text{tgt}}^{(m)} \in \mathbb{R}^D$. At $S=1$, all four targets contain the same observation; distinct residuals arise from the predictor rollout, not from distinct target horizons. Starting from $\hat{z}_{\text{tgt}}^{(0)} \equiv z_{\text{ctx}}$, a recurrent sub-horizon predictor $P_{\text{sub}}$ rolls out sequential forecasts $\hat{z}_{\text{tgt}}^{(m)} = \operatorname{LN}\bigl(P_{\text{sub}}(\hat{z}_{\text{tgt}}^{(m-1)} + \mathbf{p}_m)\bigr)$ for $m \in \{1,\dots,M-1\}$ using learned horizon embeddings $\mathbf{p}_m \in \mathbb{R}^D$, while the full-horizon anchor $P_\theta$ predicts $\hat{z}_{\text{tgt}}^{(M)} = \operatorname{LN}\bigl(P_\theta(z_{\text{ctx}} + \mathbf{p}_M)\bigr)$. With horizon weights $w_m = \frac{1}{2(M-1)}$ ($m < M$) and $w_M = \frac{1}{2}$, training minimizes multi-horizon MSE alongside a non-collapse matrix penalty $\boldsymbol{\Delta}_\gamma$:
\begin{equation}
    \mathcal{L}_{\text{Lam}} = \sum_{m=1}^M \frac{w_m}{D} \Bigl(\bigl\|\hat{z}_{\text{tgt}}^{(m)} - \operatorname{sg}\bigl(z_{\text{tgt}}^{(m)}\bigr)\bigr\|_2^2 + \bigl\|\boldsymbol{\Delta}_\gamma\bigl(\hat{z}_{\text{tgt}}^{(m)}\bigr)\bigr\|_F^2\Bigr).
    \label{eq:soliton_reynolds_loss}
\end{equation}
Here $\boldsymbol{\Delta}_\gamma(\mathbf{Z})\in\mathbb{R}^{D\times D}$ penalizes collapse without requiring $B>D$. Its diagonal entries are $\sqrt{\max(0,\gamma-\sigma_d)}$, where $\sigma_d$ uses the batch-biased variance and $10^{-4}$ stabilization ($\gamma=0.75$). Off-diagonal entries are the sample covariances $\Sigma_{dj}/\sqrt{2}$ ($d\ne j$), computed with the $B-1$ divisor. The distinct variance conventions matter for exact correspondence with the objective. At $B=16<D=32$, the measured effective rank $\operatorname{RankMe}(z_{\text{tgt}})$ rises from $11.6$ to $19.8$ \cite{garrido2023rankme}.

\subsection{Stage 3: Calibration-Stabilized Whitening, Optional Weight-Space Quenching, and 224-Dial Expansion}
\label{subsec:reynolds_expansion}
Why does collapsing a multi-horizon residual tensor into a 1D scalar score degrade real-time detection? Stacking the $M=4$ signed residuals $\mathbf{e}_m(t)=z_{\text{tgt}}^{(m)}(t)-\hat{z}_{\text{tgt}}^{(m)}(t)$ forms $\mathbf{E}(t)\in\mathbb{R}^{M\times D}$ ($d_{\text{tot}}=M\!D=128$). On weight-fitting windows, the sample residual covariance $\hat{\mathbf{\Sigma}}_{\text{res}}=\mathbf{U}\operatorname{diag}(\lambda_k)\mathbf{U}^{\top}$ defines the positive-semidefinite resolvent precision
\[
    \mathbf{\Omega}
    =\mathbf{U}\operatorname{diag}\!\left(
    \frac{\lambda_k}{(\lambda_k+10^{-3})^2}
    \right)\mathbf{U}^{\top}\in\mathcal{S}_+^{128}.
\]
Applying $\mathbf{\Omega}^{1/2}$ to residuals centered by $\bar{\bm{\mu}}_{\text{res}}\in\mathbb{R}^{128}$ gives $\mathbf{E}^{\text{white}}(t)\in\mathbb{R}^{M\times D}$, attenuating near-null modes without forcing unit variance on every coordinate. Nominal regime shifts between training and calibration can still inflate a few coordinates and dominate a scalar norm. Stage~3 of \emph{Laminar-JEPA} addresses this through two mechanisms:
\begin{enumerate}[leftmargin=*,nosep,itemsep=1pt]
    \item \emph{Optional weight-space quenching and calibration-stabilized whitening:} When nominal drift between training and calibration inflates a few coordinates, \emph{Laminar-JEPA} applies an optional zero-gradient post-training edit on training and nominal-calibration windows alone (without inverting a LayerNorm Jacobian). Each coordinate $c\in\{1,\dots,D\}$ receives a standardized instability score $r_c$ from its training-to-calibration mean shift $\delta_c$ and standard-deviation ratio $v_c$, calibration half-split shift $b_c$ and ratio $t_c\ge 1$, nonnegative lag-$\ell$ autocorrelation $\rho_c$ of the horizon-mean raw residual, and nominal peak-window trip share $h_c$:
    \begin{equation}
    \small
    \begin{aligned}
    s_c={}&0.35\delta_c+0.30(v_c-1)_++0.45b_c\\
    &+0.35(t_c-1)_++0.25\rho_c-1.50h_c,\\
    r_c={}&\frac{s_c-\operatorname{med}_j s_j}
    {1.4826\,\operatorname{med}_j|s_j-\operatorname{med}_k s_k|+10^{-4}}.
    \end{aligned}
    \label{eq:quench_score}
    \end{equation}
    For candidate top-$k$ coordinate sets $\mathcal{C}_k$ ($k\in\{4,2\}$ of $r_c$), four steps execute in order: \emph{(Q1)}~damp the $c$-th output rows and LayerNorm gains of $\{P_{\text{sub}},P_\theta\}$ by $a_c=\operatorname{clip}((1+0.5(r_c)_+)^{-1},0.35,0.85)$ and shift $\mathbf{P}_{:,c}$ by $+0.35(\bm{\mu}_{:,c}^{\text{cal}}-\bm{\mu}_{:,c}^{\text{tr}})$ (GELU and 32D LayerNorm couple coordinates, causing unselected spillover audited in Section~\ref{subsec:controlled_ablations}); \emph{(Q2)}~blend $\bar{\mu}_{\text{res}}[mD+c]$ $75\%$ toward post-edit calibration means for $c\in\mathcal{C}_k$; \emph{(Q3)}~attenuate off-diagonal precision couplings $\Omega_{m_1 D+c,m_2 D+c'}$ ($c'\ne c$) by $0.15$, project onto $\mathcal{S}_+^{MD}$ (floor $10^{-6}$), and rescale post-edit whitened calibration RMS toward $0.50$ ($c\notin\mathcal{C}_k$) and $[0.975,1.08]$ ($c\in\mathcal{C}_k$); and \emph{(Q4)}~retain the edit (preferring $k=2$ on ties) iff nominal half-split FPR improves without shrinking $K_{224}$ or $K_{128}$ below $0.92\times$ their pre-edit values, reverting otherwise. Any coordinate with $\text{RMS}_{\text{cal}}^{\text{white}}(m,c)>1.10$ then blends smoothly toward $\hat{e}_{m,c}^{\text{raw}}(t)=e_{m,c}(t)/(\text{RMS}_{\text{cal}}^{\text{raw}}(m,c)+10^{-4})$:
    \begin{equation}
    \begin{aligned}
        \tilde{e}_{m,c}(t) &= w_{m,c}\,e_{m,c}^{\text{white}}(t) + (1 - w_{m,c})\,\hat{e}_{m,c}^{\text{raw}}(t), \\
        w_{m,c} &= \frac{1}{1 + 2\max\bigl(0, \text{RMS}_{\text{cal}}^{\text{white}}(m,c) - 1.10\bigr)^2}.
    \end{aligned}
    \label{eq:stabilized_whitening}
    \end{equation}
    \item \emph{Multi-horizon expansion into $7D = 224$ dials:} Motivated algebraically by the mean-plus-fluctuation split of Reynolds averaging \cite{reynolds1895dynamical} along the discrete horizon index $m \in \{1,\dots,M\}$ of each coordinate $c \in \{1,\dots,32\}$, we decompose $\tilde{\mathbf{E}}(t) \in \mathbb{R}^{4 \times 32}$ into horizon mean $\bar{e}_c(t) = \frac{1}{M}\sum_{m=1}^M \tilde{e}_{m,c}(t)$ and zero-mean fluctuation $F_{m,c}(t) = \tilde{e}_{m,c}(t) - \bar{e}_c(t)$, forming $7D = 224$ nonnegative dials $\mathbf{d}(t) \in \mathbb{R}_{\ge 0}^{224}$:
    (a)~\emph{sub-horizon position magnitudes ($4D = 128$)} $d_{\text{pos},m,c}(t) = |\tilde{e}_{m,c}(t)|$ ($m \in \{1..4\}$);
    (b)~\emph{horizon-mean drift ($D = 32$)} $d_{\text{drift},c}(t) = |\bar{e}_c(t)|$;
    (c)~\emph{horizon fluctuation ($D = 32$)} $d_{\text{fluc},c}(t) = \sigma'_c(t) = \sqrt{\frac{1}{M}\sum_{m=1}^M F_{m,c}^2(t)}$; and
    (d)~\emph{endpoint difference ($D = 32$)} $d_{\text{accel},c}(t) = |\tilde{e}_{M,c}(t) - \tilde{e}_{1,c}(t)|$.
\end{enumerate}

Subtracting each coordinate's horizon mean splits multi-horizon error energy orthogonally into steady drift plus zero-mean fluctuation and restricts the $M\times M$ horizon Gram matrix to $\mathbf{1}_M^\perp$ (describing representation geometry, not guaranteeing detection accuracy):

\begin{proposition}[Horizon Energy Split and Spectral Concentration Bounds]
\label{prop:anti_collapse}
Let $\tilde{\mathbf{E}}(t) \in \mathbb{R}^{M \times D}$ ($M \ge 2$) denote the stabilized residual matrix with horizon-mean vector $\bar{\mathbf{e}}(t) = M^{-1}\tilde{\mathbf{E}}(t)^\top \mathbf{1}_M \in \mathbb{R}^D$ and zero-mean fluctuation matrix $\mathbf{F}(t) = \tilde{\mathbf{E}}(t) - \mathbf{1}_M \bar{\mathbf{e}}(t)^\top \in \mathbb{R}^{M \times D}$. Then:
(i)~\emph{Coordinate-wise energy orthogonality and endpoint bound:} For every $c \in \{1,\dots,D\}$, $\frac{1}{M}\sum_{m=1}^M d_{\text{pos},m,c}^2(t) = d_{\text{drift},c}^2(t) + d_{\text{fluc},c}^2(t)$ and $d_{\text{accel},c}(t) \le \sqrt{2M}\,d_{\text{fluc},c}(t)$.
(ii)~\emph{Horizon Gram matrix and spectral concentration bounds:} Define the $M\times M$ horizon Gram matrix $\mathbf{G}(t) = \mathbf{F}(t)\mathbf{F}(t)^\top \in \mathcal{S}_+^M \subset \mathbb{R}^{M\times M}$ (distinct from the $D\times D$ coordinate matrix $\mathbf{F}(t)^\top\mathbf{F}(t)$), with ordered eigenvalues $0 = \lambda_1 \le \lambda_2 \le \dots \le \lambda_M$. For any $\gamma > -1/2$ ($p = \gamma + \frac{1}{2} > 0$) and $\mathbf{F}(t) \neq \mathbf{0}$, the scale-invariant dominant-mode concentration ratio (using Lieb--Thirring power moments \cite{lieb1976inequalities}),
\begin{equation}
    R_{\text{LT}}\bigl(\mathbf{F}(t)\bigr) = \frac{\lambda_M^p}{\sum_{m=1}^M \lambda_m^p} \in \Bigl[\frac{1}{M - 1},\, 1\Bigr],
    \label{eq:lt_bounds}
\end{equation}
attains its \emph{lower bound} $\frac{1}{M-1}$ ($=\frac{1}{3}$ at $M=4$) if and only if fluctuation energy is isotropic across all $M-1$ nonzero horizon modes ($\lambda_2 = \dots = \lambda_M > 0$, requiring $D \ge M-1$), and attains its \emph{upper bound} $1$ if and only if $\mathbf{F}(t)$ has rank one ($\lambda_2 = \dots = \lambda_{M-1} = 0 < \lambda_M$). (Defining the reciprocal spread ratio $\sum_{m=1}^M \lambda_m^p / [(M-1)\lambda_M^p]$ swaps the two equality endpoints.) Empirically, scalar reweighting by $R_{\text{LT}}$ has a null effect (Section~\ref{sec:ablations}).
\end{proposition}
\begin{proof}
Since $\mathbf{F}$ has zero column sums ($F_{1,c}+\dots+F_{M,c}=0$), expanding $M^{-1}\sum_{m=1}^M (\bar{e}_c + F_{m,c})^2$ eliminates the cross term and gives the energy identity; applying $(F_{M,c}-F_{1,c})^2 \le 2(F_{M,c}^2+F_{1,c}^2) \le 2M\,d_{\text{fluc},c}^2$ proves~(i).

For~(ii), $\mathbf{F}^\top\mathbf{1}_M = \tilde{\mathbf{E}}^\top\mathbf{1}_M - \bar{\mathbf{e}}(\mathbf{1}_M^\top\mathbf{1}_M) = \mathbf{0}_D$, so $\mathbf{G}\mathbf{1}_M = \mathbf{F}(\mathbf{F}^\top\mathbf{1}_M) = \mathbf{0}_M \in \mathbb{R}^M$ and $\lambda_1 = 0$. Since $\mathbf{F} \neq \mathbf{0}$, $\operatorname{tr}(\mathbf{G}) = \|\mathbf{F}\|_F^2 > 0$ implies $\lambda_M > 0$. Because $p > 0$ and $0 \le \lambda_m^p \le \lambda_M^p$ for $m \in \{2,\dots,M\}$, summing the $M-1$ active modes gives $\lambda_M^p \le \sum_{m=2}^M \lambda_m^p \le (M-1)\lambda_M^p$. Inverting and multiplying by $\lambda_M^p > 0$ reverses the chain to $\frac{1}{M-1} \le R_{\text{LT}}(\mathbf{F}) \le 1$, with equality at $\frac{1}{M-1}$ iff $\lambda_2 = \dots = \lambda_M > 0$ and equality at $1$ iff $\lambda_2 = \dots = \lambda_{M-1} = 0 < \lambda_M$. Scaling $\mathbf{F} \mapsto \alpha\mathbf{F}$ ($\alpha \neq 0$) scales $\lambda_m \mapsto \alpha^2\lambda_m$ and cancels in (\ref{eq:lt_bounds}).
\end{proof}

\subsection{Stage 4: Zero-Collapse 224-Bit Circuit-Breaker Panel, Onset Relays, and Temporal Attribution}
\label{subsec:scoring_function}
\label{subsubsec:circuit_breaker_panel}
\label{subsubsec:causal_saliency_gating}
Rather than summing the 224 dials into a scalar norm, Stage~4 (Fig.~\ref{fig:triad_architecture}, Algorithm~\ref{alg:pipeline}) equips each dial with a binary circuit breaker calibrated on nominal data. Because all 224 dials derive from one $4 \times 32$ residual tensor, their trips are correlated by construction (median effective rank $65$ of $224$ on the development cohort, with joint trip-count variance $7.2\times$ independent trials), so alarm quorums are calibrated directly on the empirical distribution of simultaneous nominal trip counts:
\begin{enumerate}[leftmargin=*,nosep,itemsep=1pt]
    \item \emph{Per-dial standardization and nominal hijack guards:} Each dial $j \in \{1,\dots,224\}$ is standardized by its nominal calibration median $m_j$ and scaled median absolute deviation (MAD) $s_j = 1.4826\,\operatorname{MAD}(d_{j,\text{cal}}) + 10^{-4}$ into $z_j(t) = \max\bigl(0, (d_j(t) - m_j)/s_j\bigr)$. To quarantine any individual coordinate that drifts during calibration, each dial's $98.5$th-percentile tripwire $\tau_j$ is scaled by a two-factor nominal hijack guard $\gamma_j = \gamma_j^{\text{cal}}\,\gamma_j^{\text{sc}}$:
    \begin{equation}
        b_j(t) = \mathbb{I}\bigl(z_j(t) > \tau_j\bigr), \quad \tau_j = q_{0.985}(z_{j,\text{cal}})\,\gamma_j^{\text{cal}}\,\gamma_j^{\text{sc}},
        \label{eq:dial_tripwire}
    \end{equation}
    where $\gamma_j^{\text{cal}}$ is the ratio of second-half to first-half calibration 98th percentiles whenever $>1.35$ ($1$ otherwise), and $\gamma_j^{\text{sc}}$ is $(m_j + s_j)/(1.5\,s_{\text{group},j})$ whenever $>1.25$ ($1$ otherwise, relative to perspective-group median $s_{\text{group},j}$).
    \item \emph{32 fast right-edge onset relays:} To trigger rapidly when an abrupt fault enters the right edge of the $S$-step window while earlier prefixes ($m \in \{1, 2\}$) still observe nominal data, each coordinate $c \in \{1,\dots,32\}$ defines a strict edge switch $b_{p,c}^{\text{edge}}(t) = \mathbb{I}\bigl(z_{p,c}(t) > q_{0.993}(z_{p,c,\text{cal}})\,\gamma_{p,c}^{1.25}\bigr)$ and fires its onset relay $r_c(t) \in \{0, 1\}$ when a terminal-prefix trip ($m=4$) coincides with acceleration, fluctuation, or prefix $m=3$:
    \begin{equation}
        r_c(t) = b_{\text{pos},4,c}^{\text{edge}}(t) \wedge \bigl(b_{\text{accel},c}^{\text{edge}}(t) \vee b_{\text{fluc},c}^{\text{edge}}(t) \vee b_{\text{pos},3,c}^{\text{edge}}(t)\bigr).
        \label{eq:onset_relay}
    \end{equation}
    \item \emph{Causal trailing vs.\ buffered attribution and Boolean quorums:} Under \emph{causal trailing attribution}, switches on window $[t,t+S)$ emit a per-timestep decision at its right endpoint $\tau=t+S-1$ ($\mathcal{F}_\tau$-measurable, structural lag $S-1$, empirical delay $\Delta t=t_{\text{alm}}-t^*$). Under \emph{retrospective buffered attribution}, the 224-dial vector routes position votes to $[t,t+S_m)$, drift and fluctuation votes to $[t,t+S)$, and endpoint-difference votes to $[t+S_3,t+S)$; the 128-dial counter and onset relays average uniformly over $[t,t+S)$ (at $S=1$, the buffered endpoint-difference span is empty). Summing active breakers gives $N_{224}(\tau)$, $N_{128}(\tau)$, and $N_{\text{edge}}(\tau)$ (continuous ranking score $N_{\text{tot}}(\tau)=N_{224}(\tau)+N_{128}(\tau)+2N_{\text{edge}}(\tau)$ for PR-AUC). Per-timestep binary alarms $\hat{y}(\tau)\in\{0,1\}$ trigger when any counter exceeds its nominal-calibration quorum:
    \begin{equation}
    \begin{aligned}
        \hat{y}(\tau) = \mathbb{I}\bigl(&N_{224}(\tau) > K_{224} \;\vee\; N_{128}(\tau) > K_{128} \\
        &\vee\; N_{\text{edge}}(\tau) > K_{\text{edge}}\bigr),
    \end{aligned}
    \label{eq:panel_quorum}
    \end{equation}
    where $K_{\text{edge}} = \max\bigl(1.25\,q_{0.999}(N_{\text{edge},\text{cal}}),\, 2.5\bigr)$ and $(K_{224}, K_{128}) = \bigl(\max(1.12\,q_{0.995}, 5.0),\allowbreak\, \max(1.12\,q_{0.995}, 3.5)\bigr)$ on nominal trip counts.
\end{enumerate}

\subsection{Autopsy-Distilled Architecture: \emph{Laminar-JEPA-Lite}}
\label{subsec:laminar_lite}
Layer-by-layer activation tracing and component ablations of \emph{Laminar-JEPA} (Section~\ref{sec:ablations}) reveal four structural simplifications that cut trainable parameters by $35\%\text{--}52\%$ ($141.9\text{k}$ at $K < 25$; $222.3\text{k}$ at $K \ge 25$ vs.\ $295.8\text{k}$) and accelerate windowed GPU inference by $2.43\times$ ($3.14\times$ at $K < 25$) without post-training weight editing:
\begin{enumerate}[leftmargin=*,nosep,itemsep=1pt]
    \item \emph{Four-block TCN and $K \ge 25$ dyadic gating:} Because a fifth dilated block attenuates anomaly divergence on short target prefixes without raising effective rank ($R_{\text{stab}}$), \emph{Laminar-JEPA-Lite} uses four dilated blocks ($d \in \{1, 2, 4, 8\}$), compacts its temporal/statistical pooling heads to $64\text{D}$, and enables the four-shell rFFT front-end only when $K \ge 25$ (zero shell dropout; direct causal projection when $K < 25$).
    \item \emph{Parallel three-horizon single-LayerNorm predictor:} Replacing the sequential $M=4$ rollout and redundant predictor LayerNorm stack with $M=3$ nested prefixes ($S_m \in \{16, 32, 64\}$ at $S=64$, $96\times 96$ resolvent precision $\mathbf{\Omega}\in\mathcal{S}_+^{96}$) and a single parallel residual predictor, $\hat{z}_{\text{tgt}}^{(m)} = \operatorname{LN}\bigl(z_{\text{ctx}} + \mathbf{p}_m + \mathbf{W}_2\,\phi_{\text{GELU}}(\operatorname{LN}(\mathbf{W}_1(z_{\text{ctx}} + \mathbf{p}_m)))\bigr)$, removes sequential unrolling and LayerNorm cross-coupling.
    \item \emph{128-dial position-plus-drift panel and single quorum:} Retaining only the $3D=96$ sub-horizon magnitudes $|\tilde{e}_{m,c}(t)|$ and $D=32$ horizon-mean drift dials $|\bar{e}_c(t)|$ ($4D=128$ breakers total, calibrated via (\ref{eq:dial_tripwire})) eliminates redundant fluctuation/endpoint dials, onset relays, and post-hoc quenching, firing under one nominal quorum $\hat{y}_{\text{Lite}}(\tau) = \mathbb{I}\bigl(N_{128}(\tau) > \max(1.12\,q_{0.995}(N_{128,\text{cal}}),\,4.0)\bigr)$ (prefix-weighted across $[t,t+S_m)$ in buffered mode).
\end{enumerate}

\caption{Laminar-JEPA (224-Bit) and Laminar-JEPA-Lite (128-Bit) Pipeline\label{alg:pipeline}}
\scriptsize
\begin{algorithmic}[1]
\Require Nominal stream $\mathbf{X}_{\text{nom}}$, test stream $\mathbf{X}_{\text{test}}$, tier $\in\{\text{Full }(M{=}4),\text{Lite }(M{=}3)\}$, mode $\in\{\text{causal},\text{buffered}\}$.
\State \textbf{Disjoint Split \& Training:} Partition $\mathbf{X}_{\text{nom}}$ chronologically into $(\mathbf{X}_{\text{fit}},\mathbf{X}_{\text{sel}},\mathbf{X}_{\text{cal}},\text{gap}_S,\mathbf{X}_{\text{hold}})$. Standardize by $\mathbf{X}_{\text{fit}}$; train encoders and predictor(s) on stride-16 windows of $\mathbf{X}_{\text{fit}}$ via (\ref{eq:soliton_reynolds_loss}); select checkpoint on $\mathbf{X}_{\text{sel}}$ (or fixed epochs); fit $(\bar{\bm{\mu}}_{\text{res}},\mathbf{\Omega}\in\mathcal{S}_+^{MD})$ on $\mathbf{X}_{\text{fit}}$.
\If{tier $=$ Full and $q_{\text{wsp}} = 1$} \Comment{Optional gated weight-space quenching}
    \State Rank coordinates by $r_c$ (\ref{eq:quench_score}) on $(\mathbf{X}_{\text{fit}},\mathbf{X}_{\text{cal}})$; run Steps Q1--Q4 for $k\in\{4,2\}$ and retain edit iff nominal half-split and $0.92$-quorum gates pass.
\EndIf
\State \textbf{Panel Calibration:} Extract $\tilde{\mathbf{E}}_{\text{cal}}$ on $\mathbf{X}_{\text{cal}}$ via (\ref{eq:stabilized_whitening}); form $224$ dials (Full) or $128$ P+D dials (Lite); fit $(m_j,s_j,\gamma_j,\tau_j)$ via (\ref{eq:dial_tripwire}) and nominal quorum(s).
\State \textbf{Online Inference:} For stride-1 windows of $\mathbf{X}_{\text{hold}}$ or $\mathbf{X}_{\text{test}}$, compute $\tilde{\mathbf{E}}(t)\to\mathbf{d}(t)$, trip $b_j(t)$ (\ref{eq:dial_tripwire}), attribute counts to $\tau=t+S-1$ (causal) or horizon spans (buffered), and emit per-timestep $\hat{y}(\tau)\in\{0,1\}$.
\end{algorithmic}
\end{algorithm}

\noindent\emph{1D Scalar Collapse Controls:} To isolate panel gains over 1D scoring on identical backbones (Section~\ref{sec:ablations}), we compare against soft saliency pooling $g_d(\mathbf{v}) = (1-\alpha)\|\mathbf{v}\|_2 + \alpha\sqrt{d\sum_{i=1}^d w_i v_i^2}$ ($w_i \propto \exp(|v_i|/(\tau_g \sigma_{\mathbf{v}}))$, $\sigma_{\mathbf{v}} = \|\mathbf{v}\|_2/\sqrt{d} + 10^{-6}$, $\alpha = \tau_g = 0.5$). Single-horizon TS-JEPA scores $\mathcal{S}_{\text{TS}}(t) = g_D(\mathbf{e}(t))$, and 1D Multi-Horizon JEPA scores $\mathcal{S}_{\text{env}}(t) = \sqrt{\frac{1}{2}(g_{MD}(\tilde{\mathbf{e}}(t))^2 + \max_m g_D(\tilde{\mathbf{e}}_m(t))^2)}$ (modulated by $1 + \alpha_{\text{LT}}\max(0, \tanh(0.5\,z_{\text{LT}}(t)))$ for $\alpha_{\text{LT}} \in \{0.12, 0\}$), calibrated via SPOT \cite{siffer2017anomaly, balkema1974residual, grimshaw1993computing} ($u = p_{98}, q = 10^{-3}$).
